WITH abi AS (   -- analytics_bills_items: latest bill per virtual account (UBM ord=1)
    SELECT virtual_account_id, days_of_service, due_date
    FROM (
        SELECT a.virtual_account_id,
               row_number() OVER (PARTITION BY a.virtual_account_id
                                  ORDER BY a.statement_date DESC, a.bill_adjusted_end_date DESC, b.expected_next_bill_date DESC) AS ord,
               b.days_of_service,
               c.due_date
        FROM bill_management_v2.analytics_bills_items a
        JOIN bill_management_v2.bill_metas   m ON a.bill_id = m.id
        JOIN bill_management_v2.bill_items   b ON a.bill_item_id = b.id
        JOIN bill_management_v2.bill_records c ON b.bill_record_id = c.id AND c.deleted_at IS NULL
        WHERE a.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
    ) x
    WHERE x.ord = 1
),
reb AS (   -- reports_expected_bills: one expected_bill_arrival per account
    SELECT customer_id, billing_id, MIN(expected_bill_arrival) AS expected_bill_arrival
    FROM bill_management_v2.reports_expected_bills
    GROUP BY customer_id, billing_id
)
SELECT
    (CURRENT_DATE - reb.expected_bill_arrival)                          AS days_late,
    TO_CHAR(rnva.date_of_last_bill, 'DD/MM/YYYY')                       AS last_bill_invoice_date,
    rnva.billing_id                                                    AS billing_account,
    rnva.location                                                      AS location,
    rnva.account_status,
    rnva.utility_type,
    rnva.bill_type                                                     AS last_bill_type,
    TO_CHAR(abi.due_date, 'DD/MM/YYYY')                                AS last_bill_due_date,
    abi.days_of_service,
    TO_CHAR(reb.expected_bill_arrival, 'DD/MM/YYYY')                   AS expected_bill_arrival,
    TO_CHAR(abi.due_date + INTERVAL '30 days', 'DD/MM/YYYY')           AS expected_bill_due_date,
    ((abi.due_date + INTERVAL '30 days')::date - CURRENT_DATE)         AS days_to_next_expected_due_date,
    rnva.meter_id,
    rnva.service_account_id                                            AS said,
    rnva.vendor,
    l.building_type
FROM bill_management_v2.reports_new_virtual_accounts rnva
LEFT JOIN abi                                  ON abi.virtual_account_id = rnva.virtual_account_id
LEFT JOIN reb                                  ON reb.customer_id = rnva.customer_id AND reb.billing_id = rnva.billing_id
LEFT JOIN bill_management_v2.locations l       ON l.id = rnva.location_id
WHERE rnva.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
  AND rnva.account_status IN ('Open', 'Unknown')
  AND COALESCE(TRIM(rnva.billing_id), '') <> ''
  AND rnva.date_of_last_bill >= '{{ LBDateHelper.start() }}'
  AND rnva.date_of_last_bill <  '{{ LBDateHelper.end() }}'
  AND (CURRENT_DATE - reb.expected_bill_arrival) >= 0
ORDER BY days_late DESC
LIMIT 5000
