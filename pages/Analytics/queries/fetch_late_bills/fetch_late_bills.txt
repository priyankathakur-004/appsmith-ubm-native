SELECT
    e.days_late,
    TO_CHAR(e.last_bill_invoice_date, 'DD/MM/YYYY')   AS last_bill_invoice_date,
    e.billing_id                                      AS billing_account,
    l.name                                            AS location,
    e.account_status,
    e.utility                                         AS utility_type,
    TO_CHAR(e.last_bill_due_date, 'DD/MM/YYYY')       AS last_bill_due_date,
    e.average_days_of_service                         AS days_of_service,
    TO_CHAR(e.expected_bill_arrival, 'DD/MM/YYYY')    AS expected_bill_arrival,
    e.meter_id,
    e.service_account_id                              AS said,
    e.vendor
FROM bill_management_v2.reports_expected_bills_by_account_new3 e
LEFT JOIN bill_management_v2.locations l ON l.id = e.location_id
WHERE e.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
  AND e.account_status IN ('Open', 'Unknown')
  AND COALESCE(TRIM(e.billing_id), '') <> ''
  AND e.last_bill_invoice_date >= (CURRENT_DATE - INTERVAL '6 months')
  AND e.days_late >= 0
ORDER BY e.days_late DESC
LIMIT 5000
