SELECT
    MAX(e.days_late)                                                        AS days_late,
    TO_CHAR(MAX(e.last_bill_invoice_date), 'DD/MM/YYYY')                    AS last_bill_invoice_date,
    e.billing_id                                                           AS billing_account,
    string_agg(DISTINCT l.name, ', ' ORDER BY l.name)                      AS location,
    MAX(e.account_status)                                                  AS account_status,
    string_agg(DISTINCT e.utility, ', ' ORDER BY e.utility)                AS utility_type,
    string_agg(DISTINCT bm.bill_type, ', ' ORDER BY bm.bill_type)          AS last_bill_type,
    TO_CHAR(MAX(e.last_bill_due_date), 'DD/MM/YYYY')                        AS last_bill_due_date,
    MAX(e.last_bill_adj_service_end_date - e.last_bill_service_start_date)  AS days_of_service,
    TO_CHAR(MAX(e.expected_bill_arrival), 'DD/MM/YYYY')                     AS expected_bill_arrival,
    TO_CHAR(MAX(e.last_bill_due_date) + INTERVAL '30 days', 'DD/MM/YYYY')   AS expected_bill_due_date,
    ((MAX(e.last_bill_due_date) + INTERVAL '30 days')::date - CURRENT_DATE) AS days_to_next_expected_due_date,
    string_agg(DISTINCT e.meter_id, ', ' ORDER BY e.meter_id)              AS meter_id,
    e.billing_id                                                           AS said,
    string_agg(DISTINCT e.vendor, ', ' ORDER BY e.vendor)                  AS vendor
FROM bill_management_v2.reports_expected_bills_by_account e
LEFT JOIN bill_management_v2.locations l   ON l.id  = e.location_id
LEFT JOIN bill_management_v2.bill_metas bm ON bm.id = e.last_bill_id
WHERE e.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
  AND e.account_status IN ('Open', 'Unknown')
  AND COALESCE(TRIM(e.billing_id), '') <> ''
  AND e.last_bill_invoice_date >= (CURRENT_DATE - INTERVAL '6 months')
  AND e.days_late >= 0
GROUP BY e.billing_id
ORDER BY days_late DESC
LIMIT 5000
