WITH abi AS (   -- analytics_bills_items: latest bill per virtual account (UBM ord=1)
    SELECT virtual_account_id, days_of_service, due_date, last_bill_type, expected_next_bill_date
    FROM (
        SELECT a.virtual_account_id,
               row_number() OVER (PARTITION BY a.virtual_account_id
                                  ORDER BY a.statement_date DESC, a.bill_adjusted_end_date DESC, b.expected_next_bill_date DESC) AS ord,
               b.days_of_service, c.due_date, m.bill_type AS last_bill_type, b.expected_next_bill_date
        FROM bill_management_v2.analytics_bills_items a
        JOIN bill_management_v2.bill_metas   m ON a.bill_id = m.id
        JOIN bill_management_v2.bill_items   b ON a.bill_item_id = b.id
        JOIN bill_management_v2.bill_records c ON b.bill_record_id = c.id AND c.deleted_at IS NULL
        WHERE a.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
    ) x WHERE x.ord = 1
),
va AS (   -- one row per virtual account (service) for this customer
    SELECT
        rnva.billing_id                                    AS billing_account,
        rnva.date_of_last_bill,
        rnva.location, rnva.account_status, rnva.utility_type,
        rnva.meter_id, rnva.service_account_id AS said, rnva.vendor,
        l.building_type,
        abi.last_bill_type, abi.due_date, abi.days_of_service, abi.expected_next_bill_date,
        -- UBM "Days Late" counts WEEKDAYS (business days), not calendar days, between the
        -- expected bill arrival and today. Non-late (future/today) keeps the calendar value
        -- (<= 0) so the `days_late >= 0` filter below still excludes not-yet-late accounts.
        CASE
            WHEN abi.expected_next_bill_date IS NULL THEN NULL
            WHEN abi.expected_next_bill_date::date >= CURRENT_DATE
                THEN (CURRENT_DATE - abi.expected_next_bill_date::date)
            ELSE (SELECT COUNT(*)::int
                  FROM generate_series(abi.expected_next_bill_date::date + 1, CURRENT_DATE, INTERVAL '1 day') gs
                  WHERE EXTRACT(DOW FROM gs) BETWEEN 1 AND 5)
        END                                                 AS days_late
    FROM bill_management_v2.reports_new_virtual_accounts rnva
    LEFT JOIN abi                            ON abi.virtual_account_id = rnva.virtual_account_id
    LEFT JOIN bill_management_v2.locations l ON l.id = rnva.location_id
    WHERE rnva.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
      AND rnva.account_status IN ('Open', 'Unknown')
      AND COALESCE(TRIM(rnva.billing_id), '') <> ''
),
rep AS (   -- representative = the account's latest bill (one row per billing account)
    SELECT DISTINCT ON (billing_account) *
    FROM va
    ORDER BY billing_account, date_of_last_bill DESC NULLS LAST, expected_next_bill_date DESC NULLS LAST
),
agg AS (   -- combine only the LATE, in-window services per account (UBM shows e.g. "SEWER, WATER")
    SELECT billing_account,
        string_agg(DISTINCT location,      ', ' ORDER BY location)      AS locations,
        string_agg(DISTINCT utility_type,  ', ' ORDER BY utility_type)  AS utilities,
        string_agg(DISTINCT meter_id,      ', ' ORDER BY meter_id)      AS meters,
        string_agg(DISTINCT vendor,        ', ' ORDER BY vendor)        AS vendors,
        string_agg(DISTINCT building_type, ', ' ORDER BY building_type) AS building_types
    FROM va
    WHERE va.days_late >= 0
      AND va.date_of_last_bill >= '{{ BH_LBDateHelper.start() }}'
      AND va.date_of_last_bill <  '{{ BH_LBDateHelper.end() }}'
    GROUP BY billing_account
)
SELECT
    r.days_late                                                         AS days_late,
    TO_CHAR(r.date_of_last_bill, 'DD/MM/YYYY')                          AS last_bill_invoice_date,
    r.billing_account,
    a.locations                                                        AS location,
    r.account_status,
    a.utilities                                                        AS utility_type,
    r.last_bill_type,
    TO_CHAR(r.due_date, 'DD/MM/YYYY')                                  AS last_bill_due_date,
    r.days_of_service,
    TO_CHAR(r.expected_next_bill_date, 'DD/MM/YYYY')                   AS expected_bill_arrival,
    TO_CHAR(r.due_date + INTERVAL '30 days', 'DD/MM/YYYY')             AS expected_bill_due_date,
    -- Days To Next Expected Due Date is CALENDAR days in UBM (unlike Days Late which is weekdays).
    ((r.due_date + INTERVAL '30 days')::date - CURRENT_DATE)           AS days_to_next_expected_due_date,
    a.meters                                                           AS meter_id,
    r.said,
    a.vendors                                                          AS vendor,
    a.building_types                                                   AS building_type
FROM rep r
JOIN agg a ON a.billing_account = r.billing_account
WHERE r.date_of_last_bill >= '{{ BH_LBDateHelper.start() }}'
  AND r.date_of_last_bill <  '{{ BH_LBDateHelper.end() }}'
  AND r.days_late >= 0
ORDER BY r.days_late DESC
LIMIT 5000
