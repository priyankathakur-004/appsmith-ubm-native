SELECT
    m.attribute_name,
    v.attribute_value,
    v.location_id,
    v.date_month
FROM bill_management_v2.location_monthly_attributes_metadata m
INNER JOIN bill_management_v2.location_monthly_attributes_values v
    ON m.id = v.location_monthly_attributes_metadata_id
WHERE m.customer_id = {{CustomerSelect.selectedOptionValue || 76013}}
    AND m.attribute_type = 'Numeric'
    AND v.attribute_value <> ''
    {{ MADateSelect.selectedOptionValues?.filter(d => d.includes('-')).length > 0 ? `AND date_trunc('month', v.date_month::date)::date IN (${MADateSelect.selectedOptionValues.filter(d => d.includes('-')).map(d => `'${d}'`).join(",")})` : "" }}
ORDER BY m.attribute_name;
