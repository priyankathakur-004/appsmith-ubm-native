WITH acts AS (
    SELECT
        ah.bill_meta_id,
        ah.text,
        ah.created_at,
        CONCAT(u.profile->>'first_name', ' ', u.profile->>'last_name') AS user_name
    FROM bill_management_v2.activity_history ah
    JOIN bill_management_v2.bill_metas bm ON bm.id = ah.bill_meta_id
    LEFT JOIN bill_management_v2.users u ON u.id = ah.user_id
    WHERE ah.text IS NOT NULL
      AND bm.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
),
lt AS (
    SELECT DISTINCT ON (bill_meta_id)
        bill_meta_id,
        text AS last_text,
        created_at AS last_chat_date,
        user_name AS last_chat_user
    FROM acts
    ORDER BY bill_meta_id, created_at DESC
),
hist AS (
    SELECT bill_meta_id,
           STRING_AGG(text, E'\n--------------------\n' ORDER BY created_at) AS chat_history
    FROM acts
    GROUP BY bill_meta_id
),
bill_info AS (
    SELECT DISTINCT ON (bm.id)
        bm.id AS bill_meta_id,
        bm.marked_for_payment,
        abi.bill_id,
        l.name AS location
    FROM bill_management_v2.bill_metas bm
    LEFT JOIN bill_management_v2.bill_metas_bill_records bmbr ON bmbr.bill_meta_id = bm.id AND bmbr.status = 'active'
    LEFT JOIN bill_management_v2.bill_records br ON br.id = bmbr.bill_record_id
    LEFT JOIN bill_management_v2.analytics_bills_items abi ON abi.bill_record_id = br.id
    LEFT JOIN bill_management_v2.bill_items bi ON bi.bill_record_id = br.id
    LEFT JOIN bill_management_v2.bill_items_virtual_accounts biva ON biva.bill_item_id = bi.id
    LEFT JOIN bill_management_v2.locations l ON l.id = biva.location_id
    WHERE bm.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
    ORDER BY bm.id
)
SELECT
    bi.location,
    bi.bill_id,
    TO_CHAR(bi.marked_for_payment, 'MM/DD/YYYY') AS marked_for_payment,
    TO_CHAR(lt.last_chat_date, 'MM/DD/YYYY')      AS last_chat_date,
    lt.last_chat_user,
    lt.last_text                                  AS last_chat_text,
    hist.chat_history
FROM lt
JOIN bill_info bi ON bi.bill_meta_id = lt.bill_meta_id
LEFT JOIN hist ON hist.bill_meta_id = lt.bill_meta_id
ORDER BY lt.last_chat_date DESC
