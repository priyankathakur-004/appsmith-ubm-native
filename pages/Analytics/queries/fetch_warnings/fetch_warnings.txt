SELECT
    be.id            AS warning_id,
    be.message       AS bill_warning,
    be.severity,
    be.resolved,
    br.customer_id   AS cust,
    abi.bill_id      AS pear_id
FROM bill_management_v2.bill_errors be
JOIN bill_management_v2.bill_records br ON br.id = be.bill_record_id
LEFT JOIN bill_management_v2.analytics_bills_items abi ON abi.bill_record_id = br.id
WHERE br.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
ORDER BY be.severity DESC, be.id
LIMIT 100
