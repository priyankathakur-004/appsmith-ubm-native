-- What the picked customer's data holds: the date range, location count,
-- utilities with their units, and the charge categories on its bills (last
-- two years). Shown in the page header and given to the AI so it uses values
-- that exist.
SELECT
    {{ Number(appsmith.store.aiCustomer) || 0 }}         AS cust,
    TO_CHAR(MIN(m.time_period::date), 'YYYY-MM')        AS first_month,
    TO_CHAR(MAX(m.time_period::date), 'YYYY-MM')        AS last_month,
    COUNT(DISTINCT m.location_id)                       AS locations,
    STRING_AGG(DISTINCT m.utility_type, ', ')           AS utilities,
    STRING_AGG(DISTINCT m.utility_type || ' (' || COALESCE(m.total_consumption_uom, '?') || ')', ', ')
                                                        AS utility_units,
    (SELECT STRING_AGG(DISTINCT li.category, ', ')
     FROM bill_management_v2.analytics_billing_line_items li
     WHERE li.customer_id = {{ Number(appsmith.store.aiCustomer) || 0 }}
       AND li.bill_type <> 'historical'
       AND li.workflow_state = 'processed'
       AND li.statement_date >= CURRENT_DATE - INTERVAL '24 months') AS charge_categories
FROM bill_management_v2.reports_customer_monthly_usage m
WHERE m.customer_id = {{ Number(appsmith.store.aiCustomer) || 0 }};
