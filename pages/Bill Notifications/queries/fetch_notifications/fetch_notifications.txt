WITH target_user AS (   -- all persona accounts for the selected person (defaults to Helene)
    SELECT id FROM bill_management_v2.users
    WHERE CONCAT(profile->>'first_name', ' ', profile->>'last_name')
          = '{{ appsmith.store.notifUser || "Helene Piscopo" }}'
),
tagged AS (   -- one row per comment where the selected person (any persona) was tagged by name
    SELECT DISTINCT ah.id
    FROM bill_management_v2.activity_history ah
    CROSS JOIN LATERAL json_array_elements(ah.text::json->'chatTags') AS tag
    WHERE ah.type = 'manual_comment'
      AND LEFT(ah.text,1) = '{'
      AND tag->>'type' IN ('customerUser','operatorUser')
      AND (tag->>'id')::int IN (SELECT id FROM target_user)
)
SELECT
    ah.created_at                                                          AS date_time,
    ah.bill_meta_id                                                        AS bill_id,
    cs.name                                                                AS customer,
    CONCAT(author.profile->>'first_name',' ',author.profile->>'last_name') AS sender,
    ah.text::json->>'message'                                              AS message
FROM bill_management_v2.activity_history ah
JOIN tagged t ON t.id = ah.id
LEFT JOIN bill_management_v2.users author ON author.id = ah.user_id
LEFT JOIN LATERAL (
    SELECT br.customer_id
    FROM bill_management_v2.bill_metas_bill_records bmbr
    JOIN bill_management_v2.bill_records br ON br.id = bmbr.bill_record_id
    WHERE bmbr.bill_meta_id = ah.bill_meta_id AND bmbr.status = 'active'
    LIMIT 1
) bc ON true
LEFT JOIN bill_management_v2.customers_search cs ON cs.id = bc.customer_id
ORDER BY ah.created_at DESC
LIMIT 200;
