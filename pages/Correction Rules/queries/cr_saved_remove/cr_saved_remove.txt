-- Delete a saved correction rule for everyone. The row is kept, only marked,
-- and its name becomes free for a new rule.
UPDATE ubm_app.correction_rules_saved AS r
SET deleted_at = now(),
    deleted_by = input.p ->> 'by'
FROM (SELECT '{{ this.params.payload || "{}" }}'::jsonb AS p) AS input
WHERE r.id = (input.p ->> 'id')::bigint
  AND r.deleted_at IS NULL
RETURNING r.id::text AS id, r.name;
