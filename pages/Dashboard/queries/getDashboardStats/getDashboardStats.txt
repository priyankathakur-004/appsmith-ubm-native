-- Pipeline stage counts per period for the selected customer.
-- Mirrors getBillsTotalCount's joins/DISTINCT so stage counts sum to the
-- bills-grid Total. One row per period; columns follow the dashboard order.
WITH bills AS (
    SELECT DISTINCT ON (abi.bill_id, br.client_account)
           bm.workflow_state,
           bm.created_at,
           TRUE AS is_bill
    FROM bill_management_v2.bill_metas bm
    LEFT JOIN bill_management_v2.bill_metas_bill_records bmbr
        ON bmbr.bill_meta_id = bm.id AND bmbr.status = 'active'
    LEFT JOIN bill_management_v2.bill_records br
        ON br.id = bmbr.bill_record_id
    LEFT JOIN bill_management_v2.bill_items bi ON bi.bill_record_id = br.id
    LEFT JOIN bill_management_v2.bill_items_virtual_accounts biva ON biva.bill_item_id = bi.id
    LEFT JOIN bill_management_v2.locations l ON l.id = biva.location_id
    LEFT JOIN bill_management_v2.billing_line_items bli ON br.id = bli.bill_record_id
    LEFT JOIN bill_management_v2.customers_providers_pretty_name cvn
        ON cvn.code = br.vendor_code AND cvn.customer_id = br.customer_id
    LEFT JOIN bill_management_v2.analytics_bills_items abi ON br.id = abi.bill_record_id
    LEFT JOIN bill_management_v2.bills_total_amount_charges_prior btac ON abi.bill_id = btac.bill_id
    WHERE br.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
),
p (ord, label, since) AS (
    VALUES
        (1, 'Since Yesterday', (CURRENT_DATE - INTERVAL '1 day')::timestamp),
        (2, 'Last 7 Days',     (CURRENT_DATE - INTERVAL '7 day')::timestamp),
        (3, 'Last 31 Days',    (CURRENT_DATE - INTERVAL '31 day')::timestamp),
        (4, 'All-Time',        NULL::timestamp)
)
SELECT
    p.label AS period,
    COUNT(*) FILTER (WHERE b.workflow_state = 'bill_loaded')         AS loaded,
    COUNT(*) FILTER (WHERE b.workflow_state = 'integrity_check')     AS integrity,
    COUNT(*) FILTER (WHERE b.workflow_state = 'data_verification_1') AS verif1,
    COUNT(*) FILTER (WHERE b.workflow_state = 'data_verification_2') AS verif2,
    COUNT(*) FILTER (WHERE b.workflow_state = 'data_audit_1')        AS audit1,
    COUNT(*) FILTER (WHERE b.workflow_state = 'data_audit_2')        AS audit2,
    COUNT(*) FILTER (WHERE b.workflow_state = 'processed')           AS processed,
    COUNT(b.is_bill) AS total
FROM p
LEFT JOIN bills b ON (p.since IS NULL OR b.created_at >= p.since)
GROUP BY p.ord, p.label
ORDER BY p.ord;
