-- Total rows behind a clicked dashboard cell (for pager).
WITH bills AS (
    SELECT DISTINCT ON (abi.bill_id)
           COALESCE(cvn.pretty_name, br.vendor_code)    AS "Vendor",
           TO_CHAR(br.statement_date, 'MM/DD/YYYY')     AS "Invoice Date",
           br.client_account                            AS "Billing ID",
           abi.bill_id                                  AS "Bill Id",
           TO_CHAR(bi.start_date, 'MM/DD/YYYY')         AS "Billing Start Date",
           TO_CHAR(bi.end_date, 'MM/DD/YYYY')           AS "Billing End Date",
           btac.total_charges                           AS "Current Charges",
           btac.total_amount                            AS "Total Amount Due",
           btac.total_amount                            AS "Pay Amount",
           bm.workflow_state                            AS "Status",
           bm.created_at                                AS created_at,
           br.statement_date                            AS inv_date,
           bm.bill_type                                 AS bill_type,
           br.vendor_code                               AS vendor_code,
           bi.commodity                                 AS utility,
           bm.deleted_at                                AS deleted_at
    FROM bill_management_v2.bill_metas bm
    LEFT JOIN bill_management_v2.bill_metas_bill_records bmbr ON bmbr.bill_meta_id = bm.id AND bmbr.status = 'active'
    LEFT JOIN bill_management_v2.bill_records br ON br.id = bmbr.bill_record_id
    LEFT JOIN bill_management_v2.bill_items bi ON bi.bill_record_id = br.id
    LEFT JOIN bill_management_v2.customers_providers_pretty_name cvn ON cvn.code = br.vendor_code AND cvn.customer_id = br.customer_id
    JOIN bill_management_v2.analytics_bills_items abi ON br.id = abi.bill_record_id AND abi.bill_id IS NOT NULL
    LEFT JOIN bill_management_v2.bills_total_amount_charges_prior btac ON abi.bill_id = btac.bill_id
    WHERE br.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
    ORDER BY abi.bill_id, bm.id DESC
)
SELECT COUNT(*) AS total_rows
FROM bills
WHERE ('{{ appsmith.store.drillStage || "all" }}' = 'all' OR "Status" = '{{ appsmith.store.drillStage }}')
  AND (
        '{{ appsmith.store.drillPeriod || "All-Time" }}' = 'All-Time'
     OR ('{{ appsmith.store.drillPeriod }}' = 'Since Yesterday' AND created_at >= CURRENT_DATE - INTERVAL '1 day')
     OR ('{{ appsmith.store.drillPeriod }}' = 'Last 7 Days'     AND created_at >= CURRENT_DATE - INTERVAL '7 day')
     OR ('{{ appsmith.store.drillPeriod }}' = 'Last 31 Days'    AND created_at >= CURRENT_DATE - INTERVAL '31 day')
      )
  AND (
        '{{ (appsmith.store.drillSearch || "").replace(/'/g,"''") }}' = ''
     OR "Billing ID"  ILIKE '%{{ (appsmith.store.drillSearch || "").replace(/'/g,"''") }}%'
     OR "Vendor"      ILIKE '%{{ (appsmith.store.drillSearch || "").replace(/'/g,"''") }}%'
     OR "Bill Id"::text ILIKE '%{{ (appsmith.store.drillSearch || "").replace(/'/g,"''") }}%'
      )
  AND ('{{ appsmith.store.drillVendor || "" }}'   = '' OR vendor_code = '{{ appsmith.store.drillVendor }}')
  AND ('{{ appsmith.store.drillBillType || "" }}' = '' OR bill_type   = '{{ appsmith.store.drillBillType }}')
  AND ('{{ appsmith.store.drillUtility || "" }}'  = '' OR utility     = '{{ appsmith.store.drillUtility }}')
  AND ('{{ appsmith.store.drillCostFrom || "" }}' = '' OR "Total Amount Due" >= {{ appsmith.store.drillCostFrom || 0 }})
  AND ('{{ appsmith.store.drillCostTo || "" }}'   = '' OR "Total Amount Due" <= {{ appsmith.store.drillCostTo || 0 }})
  AND (NULLIF('{{ appsmith.store.drillInvFrom || "" }}', '') IS NULL     OR inv_date   >= NULLIF('{{ appsmith.store.drillInvFrom || "" }}', '')::timestamp)
  AND (NULLIF('{{ appsmith.store.drillInvTo || "" }}', '') IS NULL       OR inv_date   <= NULLIF('{{ appsmith.store.drillInvTo || "" }}', '')::timestamp)
  AND (NULLIF('{{ appsmith.store.drillLoadedFrom || "" }}', '') IS NULL  OR created_at >= NULLIF('{{ appsmith.store.drillLoadedFrom || "" }}', '')::timestamp)
  AND (NULLIF('{{ appsmith.store.drillLoadedTo || "" }}', '') IS NULL    OR created_at <= NULLIF('{{ appsmith.store.drillLoadedTo || "" }}', '')::timestamp)
  AND ('{{ appsmith.store.drillIncludeDeleted }}' = 'true' OR deleted_at IS NULL);
