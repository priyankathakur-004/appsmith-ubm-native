WITH bill_info AS (
    SELECT DISTINCT ON (abi.bill_id)
        abi.bill_id,
        bm.marked_for_payment,
        bm.workflow_state,
        l.name AS location
    FROM bill_management_v2.analytics_bills_items abi
    JOIN bill_management_v2.bill_records br ON br.id = abi.bill_record_id
    LEFT JOIN bill_management_v2.bill_metas_bill_records bmbr ON bmbr.bill_record_id = br.id AND bmbr.status = 'active'
    LEFT JOIN bill_management_v2.bill_metas bm ON bm.id = bmbr.bill_meta_id
    LEFT JOIN bill_management_v2.bill_items bi ON bi.bill_record_id = br.id
    LEFT JOIN bill_management_v2.bill_items_virtual_accounts biva ON biva.bill_item_id = bi.id
    LEFT JOIN bill_management_v2.locations l ON l.id = biva.location_id
    WHERE br.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
      AND abi.bill_id IN (
          SELECT DISTINCT ah.bill_meta_id
          FROM bill_management_v2.activity_history ah
          WHERE ah.type = 'manual_comment'
      )
    ORDER BY abi.bill_id
),
acts AS (   -- chat comments only; newer comments store JSON text -> extract the "message"
    SELECT
        ah.bill_meta_id AS bill_id,
        CASE WHEN LEFT(ah.text, 1) = '{' THEN ah.text::json->>'message' ELSE ah.text END AS text,
        ah.created_at,
        CONCAT(u.profile->>'first_name', ' ', u.profile->>'last_name') AS user_name
    FROM bill_management_v2.activity_history ah
    LEFT JOIN bill_management_v2.users u ON u.id = ah.user_id
    WHERE ah.type = 'manual_comment'
      AND ah.bill_meta_id IN (SELECT bill_id FROM bill_info)
),
lt AS (
    SELECT DISTINCT ON (bill_id)
        bill_id,
        text       AS last_text,
        created_at AS last_chat_date,
        user_name  AS last_chat_user
    FROM acts
    ORDER BY bill_id, created_at DESC
),
hist AS (
    SELECT bill_id,
           STRING_AGG(text, E'\n--------------------\n' ORDER BY created_at) AS chat_history
    FROM acts
    GROUP BY bill_id
),
dims AS (   -- all utilities / locations / account statuses per bill (for the page filters, applied JS-side)
    SELECT abi.bill_id,
        STRING_AGG(DISTINCT l.name, ', ')               AS locations_all,
        STRING_AGG(DISTINCT bi.commodity, ', ')         AS utilities_all,
        STRING_AGG(DISTINCT rnva.account_status, ', ')  AS statuses_all
    FROM bill_management_v2.analytics_bills_items abi
    JOIN bill_management_v2.bill_records br ON br.id = abi.bill_record_id
    LEFT JOIN bill_management_v2.bill_items bi ON bi.bill_record_id = br.id
    LEFT JOIN bill_management_v2.bill_items_virtual_accounts biva ON biva.bill_item_id = bi.id
    LEFT JOIN bill_management_v2.locations l ON l.id = biva.location_id
    LEFT JOIN bill_management_v2.reports_new_virtual_accounts rnva ON rnva.virtual_account_id = biva.virtual_account_id
    WHERE br.customer_id = {{ CustomerSelect.selectedOptionValue || 76013 }}
      AND abi.bill_id IN (SELECT bill_id FROM bill_info)
    GROUP BY abi.bill_id
)
SELECT
    bi.location,
    bi.bill_id,
    bi.workflow_state,
    TO_CHAR(bi.marked_for_payment, 'MM/DD/YYYY') AS marked_for_payment,
    TO_CHAR(lt.last_chat_date, 'MM/DD/YYYY')      AS last_chat_date,
    lt.last_chat_user,
    lt.last_text                                  AS last_chat_text,
    hist.chat_history,
    d.locations_all,
    d.utilities_all,
    d.statuses_all
FROM lt
JOIN bill_info bi ON bi.bill_id = lt.bill_id
LEFT JOIN hist ON hist.bill_id = lt.bill_id
LEFT JOIN dims d ON d.bill_id = lt.bill_id
ORDER BY lt.last_chat_date DESC
LIMIT 1000
