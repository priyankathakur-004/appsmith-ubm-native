SELECT
    m.customer_id,
    m.time_period,
    m.utility_type,
    m.bill_type,
    m.location_id,
    m.location_description,
    m.client_account AS service_account,
    m.meter_serial AS meter,
    COALESCE(
        NULLIF(btrim(cvpn.pretty_name), ''),
        NULLIF(btrim(pvn.pretty_name), ''),
        v.vendor_code,
        'Unknown'
    ) AS vendor_name,
    CASE
        WHEN m.bill_type = 'Supply Only'
        THEN m.total_gen_consumption
        ELSE m.total_consumption
    END AS consumption,
    m.total_consumption_uom,
    m.total_charges
FROM bill_management_v2.reports_customer_monthly_usage m
INNER JOIN bill_management_v2.locations l
    ON l.id = m.location_id
LEFT JOIN LATERAL (
    SELECT amf.vendor_code
    FROM bill_management_v2.analytics_monthly_feed amf
    WHERE amf.customer_id = m.customer_id
      AND amf.location_id = m.location_id
      AND amf.time_period = m.time_period
      AND amf.utility_type = m.utility_type
    LIMIT 1
) v ON TRUE
LEFT JOIN bill_management_v2.providers_pretty_name pvn
    ON pvn.code = v.vendor_code
LEFT JOIN bill_management_v2.customers_vendors_pretty_name cvpn
    ON cvpn.customer_id = m.customer_id
    AND cvpn.vendor_id = pvn.vendor_id

WHERE m.customer_id = {{CustomerSelect.selectedOptionValue || 76013 }}

{{ EA_Filters.treeMonthSql() }}

{{ EAUtilityTypeSelect.selectedOptionValue && EAUtilityTypeSelect.selectedOptionValue !== 'All' ?
`AND m.utility_type = '${EAUtilityTypeSelect.selectedOptionValue}'`
: "" }}

{{ EABillTypeSelect.selectedOptionValue && EABillTypeSelect.selectedOptionValue !== 'All' ?
`AND m.bill_type = '${EABillTypeSelect.selectedOptionValue}'`
: "" }}

{{ EALocationSelect.selectedOptionValue && EALocationSelect.selectedOptionValue !== 'All' ?
`AND m.location_id = ${EALocationSelect.selectedOptionValue}`
: "" }}

{{ EALocationAttrSelect.selectedOptionValue && EALocationAttrSelect.selectedOptionValue !== 'All' ?
`AND EXISTS (
    SELECT 1
    FROM jsonb_array_elements(m.location_attributes->'custom_attributes') attr
    WHERE attr->>'id' = '${EALocationAttrSelect.selectedOptionValue}'
    ${EAAttrChoiceSelect.selectedOptionValue && EAAttrChoiceSelect.selectedOptionValue !== 'All'
      ? `AND attr->>'value' = '${EAAttrChoiceSelect.selectedOptionValue}'`
      : "" }
)`
: "" }}

ORDER BY m.location_description, m.time_period DESC
LIMIT 50000
