-- Communication Error Log (email delivery / notification errors).
-- TODO: confirm the real table + column names; this is a best-guess placeholder.
SELECT
    ce.id                                          AS id,
    TO_CHAR(ce.created_at,        'MM/DD/YYYY')     AS initial_creation_date,
    ce.status                                       AS status,
    ce.type                                         AS type,
    TO_CHAR(ce.status_changed_at, 'MM/DD/YYYY')     AS status_change_date,
    cs.name                                         AS customer,
    ce.module                                       AS module,
    ce.email_address                                AS email_address
FROM bill_management_v2.communication_errors ce
LEFT JOIN bill_management_v2.customers_search cs ON cs.id = ce.customer_id
WHERE 1 = 1
    {{ appsmith.store.comm_customerId ? "AND ce.customer_id = " + appsmith.store.comm_customerId : "" }}
    {{ appsmith.store.comm_status ? "AND ce.status = '" + appsmith.store.comm_status + "'" : "" }}
    {{ (appsmith.store.comm_types && appsmith.store.comm_types.length) ? "AND ce.type IN (" + appsmith.store.comm_types.map(t => "'" + t + "'").join(",") + ")" : "" }}
    {{ appsmith.store.comm_module ? "AND ce.module = '" + appsmith.store.comm_module + "'" : "" }}
    {{ appsmith.store.comm_email ? "AND ce.email_address = '" + appsmith.store.comm_email + "'" : "" }}
    {{ appsmith.store.comm_initCreate ? "AND ce.created_at::date = '" + appsmith.store.comm_initCreate + "'" : "" }}
    {{ appsmith.store.comm_statusChange ? "AND ce.status_changed_at::date = '" + appsmith.store.comm_statusChange + "'" : "" }}
ORDER BY ce.created_at DESC NULLS LAST
LIMIT 5000;
