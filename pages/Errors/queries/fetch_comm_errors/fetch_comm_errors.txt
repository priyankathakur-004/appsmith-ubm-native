-- Communication Error Log (email delivery / notification errors).
-- TODO: confirm the real table + column names; this is a best-guess placeholder.
SELECT
    ce.id                                          AS id,
    TO_CHAR(ce.created_at,        'MM/DD/YYYY')     AS initial_creation_date,
    ce.status                                       AS status,
    ce.type                                         AS type,
    TO_CHAR(ce.status_changed_at, 'MM/DD/YYYY')     AS status_change_date,
    cs.name                                         AS customer,
    ce.module                                       AS module,
    ce.email_address                                AS email_address
FROM bill_management_v2.communication_errors ce
LEFT JOIN bill_management_v2.customers_search cs ON cs.id = ce.customer_id
WHERE 1 = 1
    {{ CommCustomer.selectedOptionValue ? "AND ce.customer_id = " + CommCustomer.selectedOptionValue : "" }}
    {{ CommStatus.selectedOptionValue ? "AND ce.status = '" + CommStatus.selectedOptionValue + "'" : "" }}
    {{ (CommType.selectedOptionValues && CommType.selectedOptionValues.length) ? "AND ce.type IN (" + CommType.selectedOptionValues.map(t => "'" + t + "'").join(",") + ")" : "" }}
    {{ CommModule.selectedOptionValue ? "AND ce.module = '" + CommModule.selectedOptionValue + "'" : "" }}
    {{ CommEmail.selectedOptionValue ? "AND ce.email_address = '" + CommEmail.selectedOptionValue + "'" : "" }}
    {{ CommInitDate.selectedDate ? "AND ce.created_at::date = '" + moment(CommInitDate.selectedDate).format('YYYY-MM-DD') + "'" : "" }}
    {{ CommChangeDate.selectedDate ? "AND ce.status_changed_at::date = '" + moment(CommChangeDate.selectedDate).format('YYYY-MM-DD') + "'" : "" }}
ORDER BY ce.created_at DESC NULLS LAST
LIMIT 5000;
