-- Location options for the Bill Errors filter.
--
-- Read from the locations dimension, not the monthly usage feed. The feed has a
-- row per account per month, so grouping it just to list locations scanned
-- millions of rows and timed out. locations has one row per location, and
-- locations.id is the same key the feed stores as location_id - which is how the
-- bill-health warnings join the two - so the ids still match what the filter
-- compares against.
--
-- Customer is optional. Search runs server-side over the description or the id,
-- so a long list never has to reach the browser; reading the widget's own
-- filterText here is the documented pattern and is not circular, because the
-- query runs from onFilterUpdate rather than from the options binding.
SELECT l.id                                                          AS value,
       COALESCE(NULLIF(btrim(ld.location_description::text), ''),
                'Location ' || l.id::text)                           AS label
FROM bill_management_v2.locations l
LEFT JOIN bill_management_v2.location_detail ld ON ld.location_id = l.id
WHERE ('{{ ValCustomer.selectedOptionValue || "" }}' = ''
       OR l.customer_id = {{ ValCustomer.selectedOptionValue || 0 }})
  AND ('{{ (ValLocation.filterText || "").trim() }}' = ''
       OR ld.location_description::text ILIKE '%{{ (ValLocation.filterText || "").trim().split("'").join("''") }}%'
       OR l.id::text                    ILIKE '%{{ (ValLocation.filterText || "").trim().split("'").join("''") }}%')
ORDER BY label
LIMIT 1000;
