-- Location options for the Bill Errors filter.
--
-- Sourced from the same table the filter matches on: bills reach a location
-- through the monthly usage feed, so listing locations from anywhere else can
-- offer one that no bill will ever match.
--
-- Server-side search. The list is capped, so rather than shipping every location
-- to the browser this re-queries on what the user types - matching the
-- description or the id - and narrows the cap around that. Reading the widget's
-- own filterText here is the documented pattern and is not circular: the query
-- runs from onFilterUpdate, not from the options binding.
--
-- One row per location_id, guaranteed: a DISTINCT over (id, description) would
-- emit the same id twice whenever the description varies between rows, and a
-- dropdown cannot use duplicate values. Cast to text before trimming, since the
-- column type is not guaranteed.
SELECT rcmu.location_id                                              AS value,
       MAX(COALESCE(NULLIF(btrim(rcmu.location_description::text), ''),
                    'Location ' || rcmu.location_id::text))          AS label
FROM bill_management_v2.reports_customer_monthly_usage rcmu
WHERE rcmu.location_id IS NOT NULL
  -- Customer is optional: with none picked this lists locations across all of them.
  AND ('{{ ValCustomer.selectedOptionValue || "" }}' = ''
       OR rcmu.customer_id = {{ ValCustomer.selectedOptionValue || 0 }})
  AND ('{{ (ValLocation.filterText || "").trim() }}' = ''
       OR rcmu.location_description::text ILIKE '%{{ (ValLocation.filterText || "").trim().split("'").join("''") }}%'
       OR rcmu.location_id::text          ILIKE '%{{ (ValLocation.filterText || "").trim().split("'").join("''") }}%')
GROUP BY rcmu.location_id
ORDER BY label
LIMIT 200;
