-- Bill Validation Log — the bill errors behind one clicked (code, category) pair.
-- Server-side paginated: one page of rows per request, with no ceiling on how
-- far you can page. Status and search are applied here too - with paging,
-- filtering in the browser would only ever filter the page on screen.
--
-- One row more than the page is fetched so the widget knows whether a next page
-- exists without a second COUNT over the same rows; it shows the page and keeps
-- the extra only as the answer to "is there more".
--
-- Deep paging with OFFSET makes Postgres walk and discard the skipped rows, so
-- this gets slower the further in you go. That is tolerable for an investigation
-- view people use a page or two of; it would not be for a report read end to end.
SELECT
    abi.bill_id                              AS "Bill Id",
    cs.name                                  AS "Customer",
    br.client_account                        AS "Billing ID",
    -- Same name resolution as the vendor filter, so the column and the
    -- dropdown agree instead of one showing a code and the other a name.
    COALESCE(NULLIF(btrim(cvn.pretty_name::text), ''),
             br.vendor_code)                 AS "Vendor",
    TO_CHAR(br.statement_date, 'MM/DD/YYYY') AS "Invoice Date",
    be.severity                              AS severity_code,
    -- Reported exactly as stored; the schema has no label lookup for the ladder.
    be.severity::text                        AS "Severity",
    be.message                               AS "Message",
    -- data carries {commodity, account_code, meter_serial}; read it instead of
    -- regexing the message the way the warnings helper has to.
    be.data ->> 'commodity'                  AS "Utility",
    be.data ->> 'account_code'               AS "Account",
    be.data ->> 'meter_serial'               AS "Meter",
    be.resolved                              AS "Resolved",
    TO_CHAR(be.created_at, 'MM/DD/YYYY')     AS "Raised"
FROM bill_management_v2.bill_errors be
JOIN bill_management_v2.bill_records br ON br.id = be.bill_record_id
LEFT JOIN bill_management_v2.customers_search cs ON cs.id = br.customer_id
LEFT JOIN bill_management_v2.customers_providers_pretty_name cvn
       ON cvn.code = br.vendor_code AND cvn.customer_id = br.customer_id
LEFT JOIN bill_management_v2.analytics_bills_items abi
       ON abi.bill_record_id = br.id AND abi.bill_id IS NOT NULL
-- Same optional filters as the catalogue; bill_records is already joined here
-- for its columns, so they apply directly rather than through a semi-join.
WHERE ('{{ ValCustomer.selectedOptionValue || "" }}' = ''
       OR br.customer_id = {{ ValCustomer.selectedOptionValue || 0 }})
  AND ('{{ ValVendor.selectedOptionValue || "" }}' = ''
       OR br.vendor_code = '{{ ValVendor.selectedOptionValue || "" }}')
  AND ('{{ (ValAccount.text || "").trim() }}' = ''
       OR br.client_account ILIKE '%{{ (ValAccount.text || "").split("'").join("''") }}%')
  AND ('{{ ValLocation.selectedOptionValue || "" }}' = ''
       OR br.client_account IN (
              SELECT rcmu.client_account
              FROM bill_management_v2.reports_customer_monthly_usage rcmu
              WHERE rcmu.location_id = {{ ValLocation.selectedOptionValue || 0 }}))
  AND be.validation_check_id = {{ appsmith.store.valCode || 0 }}
  AND be.category = '{{ appsmith.store.valCategory || "" }}'
  AND be.created_at >= COALESCE(
          NULLIF('{{ ValDateFrom.formattedDate || "" }}', '')::timestamp,
          (CURRENT_DATE - INTERVAL '1 year')::timestamp)
  AND (NULLIF('{{ ValDateTo.formattedDate || "" }}', '') IS NULL
       OR be.created_at < NULLIF('{{ ValDateTo.formattedDate || "" }}', '')::timestamp
                          + INTERVAL '1 day')
  AND ('{{ appsmith.store.valBillStatus || "all" }}' = 'all'
       OR ('{{ appsmith.store.valBillStatus || "all" }}' = 'open'     AND be.resolved IS FALSE)
       OR ('{{ appsmith.store.valBillStatus || "all" }}' = 'resolved' AND be.resolved))
  AND ('{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}' = ''
       OR be.message               ILIKE '%{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}%'
       OR br.client_account        ILIKE '%{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}%'
       OR COALESCE(cs.name, '')    ILIKE '%{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}%'
       OR abi.bill_id::text        ILIKE '%{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}%'
       OR COALESCE(be.data ->> 'account_code', '')  ILIKE '%{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}%'
       OR COALESCE(be.data ->> 'meter_serial', '')  ILIKE '%{{ (appsmith.store.valBillSearch || "").trim().split("'").join("''") }}%')
ORDER BY be.created_at DESC, be.id DESC
LIMIT {{ (appsmith.store.valBillPageSize || 50) + 1 }}
OFFSET {{ ((appsmith.store.valBillPageNo || 1) - 1) * (appsmith.store.valBillPageSize || 50) }};
