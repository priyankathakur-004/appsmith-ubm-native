-- Bill Validation Log — the bill errors behind one clicked (code, category) pair.
-- Capped at the most recent 1000 instances; the modal paginates that slice and
-- says so, rather than pulling an unbounded number of rows for a read-only view.
-- bill_errors has no bill_id; it is reached through analytics_bills_items, the
-- same hop the per-bill error panel already uses.
SELECT
    abi.bill_id                              AS "Bill Id",
    br.client_account                        AS "Billing ID",
    br.vendor_code                           AS "Vendor",
    TO_CHAR(br.statement_date, 'MM/DD/YYYY') AS "Invoice Date",
    be.severity                              AS severity_code,
    -- Reported exactly as stored; the schema has no label lookup for the ladder.
    be.severity::text                        AS "Severity",
    be.message                               AS "Message",
    -- data carries {commodity, account_code, meter_serial}; read it instead of
    -- regexing the message the way the warnings helper has to.
    be.data ->> 'commodity'                  AS "Utility",
    be.data ->> 'account_code'               AS "Account",
    be.data ->> 'meter_serial'               AS "Meter",
    be.resolved                              AS "Resolved",
    TO_CHAR(be.created_at, 'MM/DD/YYYY')     AS "Raised"
FROM bill_management_v2.bill_errors be
JOIN bill_management_v2.bill_records br ON br.id = be.bill_record_id
LEFT JOIN bill_management_v2.analytics_bills_items abi
       ON abi.bill_record_id = br.id AND abi.bill_id IS NOT NULL
WHERE br.customer_id = {{ ValCustomer.selectedOptionValue || 76013 }}
  AND be.validation_check_id = {{ appsmith.store.valCode || 0 }}
  AND be.category = '{{ appsmith.store.valCategory || "" }}'
  AND be.created_at >= COALESCE(
          NULLIF('{{ ValDateFrom.formattedDate || "" }}', '')::timestamp,
          (CURRENT_DATE - INTERVAL '1 year')::timestamp)
  AND (NULLIF('{{ ValDateTo.formattedDate || "" }}', '') IS NULL
       OR be.created_at < NULLIF('{{ ValDateTo.formattedDate || "" }}', '')::timestamp
                          + INTERVAL '1 day')
ORDER BY be.created_at DESC, be.id DESC
LIMIT 1000;
