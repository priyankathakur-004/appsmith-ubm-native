-- Bill Validation Log — header tiles. One row per pipeline stage for the selected
-- customer and date window, plus the unresolvable backlog that dominates the page.
-- Scoping matches the catalogue query; keep the two WHERE clauses in step.
WITH scoped AS (
    SELECT be.validation_check_id AS code,
           be.resolved            AS resolved,
           be.bill_record_id      AS bill_record_id
    FROM bill_management_v2.bill_errors be
    JOIN bill_management_v2.bill_records br ON br.id = be.bill_record_id
    WHERE br.customer_id = {{ ValCustomer.selectedOptionValue || 76013 }}
      AND be.created_at >= COALESCE(
              NULLIF('{{ ValDateFrom.formattedDate || "" }}', '')::timestamp,
              (CURRENT_DATE - INTERVAL '1 year')::timestamp)
      AND (NULLIF('{{ ValDateTo.formattedDate || "" }}', '') IS NULL
           OR be.created_at < NULLIF('{{ ValDateTo.formattedDate || "" }}', '')::timestamp
                              + INTERVAL '1 day')
),
tagged AS (
    SELECT s.resolved,
           s.bill_record_id,
           COALESCE(d.workflow_state, 'unmapped') AS workflow_state,
           COALESCE(d.resolvable, FALSE)          AS resolvable
    FROM scoped s
    LEFT JOIN bill_management_v2.bill_validation_definitions d
           ON d.id = s.code AND d.deleted_at IS NULL
),
stages (ord, state, label) AS (
    VALUES (1, 'integrity_check',     'Integrity Check'),
           (2, 'data_verification_1', 'Data Verification I'),
           (3, 'data_verification_2', 'Data Verification II'),
           (4, 'data_audit_1',        'Data Audit I'),
           (5, 'data_audit_2',        'Data Audit II'),
           (6, 'unmapped',            'Unmapped')
)
SELECT
    st.label                                                      AS "Stage",
    COUNT(t.resolved)                                             AS "Total",
    COUNT(*) FILTER (WHERE t.resolved IS FALSE)                   AS "Open",
    COUNT(*) FILTER (WHERE t.resolved)                            AS "Resolved",
    -- Open errors on checks the pipeline offers no way to resolve: the real backlog.
    COUNT(*) FILTER (WHERE t.resolved IS FALSE AND t.resolvable IS FALSE)
                                                                  AS "Open, Not Resolvable",
    COUNT(DISTINCT t.bill_record_id)                              AS "Bills Affected",
    ROUND(100.0 * COUNT(*) FILTER (WHERE t.resolved)
          / NULLIF(COUNT(t.resolved), 0), 1)                      AS "Resolved %"
FROM stages st
LEFT JOIN tagged t ON t.workflow_state = st.state
GROUP BY st.ord, st.label
ORDER BY st.ord;
