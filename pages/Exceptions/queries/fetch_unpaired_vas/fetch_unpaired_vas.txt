-- Unpaired Virtual Accounts, grouped by site & utility.
-- Leads with the fast base table virtual_accounts (indexed on customer_id) and
-- joins out to the location, instead of the slow virtual_accounts_locations view.
-- Only supply/distribution accounts need pairing (Full Service excluded).
-- "paired" comes from virtual_accounts_pairing_id; the widget's No Pair / Pair
-- toggle filters on it (default: No Pair).
-- TODO(confirm): the Site label column (location_description) and whether last_bill
-- should be surfaced (it lives on the slow view, so it is blank here for now).
SELECT
    vap.location_id                                  AS location_id,
    COALESCE(NULLIF(ld.location_description, ''), 'Site ' || vap.location_id) AS site,
    va.commodity                                     AS utility,
    REPLACE(COALESCE(va.account_code, ''), '-', '')  AS billing_account_id,
    va.vendor_code                                   AS vendor,
    va.account_code                                  AS billing_id,
    va.client_account                                AS service_account_id,
    va.meter_serial                                  AS meter_id,
    va.bill_type                                     AS type,
    ''                                               AS last_bill,
    va.pair_status                                   AS pair_status,
    (va.virtual_accounts_pairing_id IS NOT NULL)     AS paired,
    va.id                                            AS va_id
FROM bill_management_v2.virtual_accounts va
LEFT JOIN bill_management_v2.virtual_accounts_locations_pairing vap
       ON vap.virtual_account_id = va.id
LEFT JOIN bill_management_v2.location_detail ld
       ON ld.location_id = vap.location_id
WHERE va.customer_id = {{ appsmith.store.selectedCustomerId || 76013 }}
  AND va.bill_type IN ('Supply Only', 'Distribution Only')
  -- Only the actionable statuses are shown by the No Pair / Pair toggle;
  -- excluding 'No Pair'/blank here cuts rows and join work.
  AND va.pair_status IN ('Need Pair', 'Pair')
ORDER BY vap.location_id, va.commodity, va.id
LIMIT 2000;
