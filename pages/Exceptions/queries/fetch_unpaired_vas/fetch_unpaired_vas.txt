-- Unpaired Virtual Accounts, grouped by site & utility.
-- Uses reports_new_virtual_accounts (the fast per-customer reporting table the
-- Late Bills feature also uses) instead of the virtual_accounts_locations view,
-- which was too slow to scan.
-- TODO(confirm): (1) the exact "No Pair" filter (account_status is a best-effort
-- proxy), (2) the Site label source, and (3) the "Type" column source
-- (Supply Only / Distribution Only was on the slow view; blank here for now).
SELECT
    rnva.location_id                                  AS location_id,
    COALESCE(NULLIF(rnva.location, ''), 'Site ' || rnva.location_id) AS site,
    rnva.utility_type                                 AS utility,
    rnva.billing_id                                   AS billing_account_id,
    rnva.vendor                                       AS vendor,
    rnva.billing_id                                   AS billing_id,
    rnva.service_account_id                           AS service_account_id,
    rnva.meter_id                                     AS meter_id,
    rnva.account_status                               AS type,
    TO_CHAR(rnva.date_of_last_bill, 'MM/DD/YYYY')     AS last_bill,
    rnva.virtual_account_id                           AS va_id
FROM bill_management_v2.reports_new_virtual_accounts rnva
WHERE rnva.customer_id = {{ appsmith.store.selectedCustomerId || 76013 }}
  AND rnva.account_status IN ('Open', 'Unknown')
ORDER BY rnva.location_id, rnva.utility_type, rnva.virtual_account_id
LIMIT 2000;
