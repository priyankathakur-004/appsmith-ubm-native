-- Unpaired Virtual Accounts: VAs mapped to a site/location, grouped by site & utility.
-- These are the supply/distribution accounts pending pairing.
-- TODO(confirm): (1) the exact "No Pair" filter (which column marks an account paired),
-- and (2) the Site label source ("Site 100" — location number vs description).
SELECT
    va.location_id                            AS location_id,
    COALESCE(NULLIF(ld.location_description, ''), 'Site ' || va.location_id) AS site,
    va.commodity                              AS utility,
    va.account_code                           AS billing_account_id,
    v.pretty_name                             AS vendor,
    va.account_code                           AS billing_id,
    va.client_account                         AS service_account_id,
    va.meter_serial                           AS meter_id,
    va.bill_type                              AS type,
    TO_CHAR(va.last_bill, 'MM/DD/YYYY')       AS last_bill,
    va.id                                     AS va_id
FROM bill_management_v2.virtual_accounts_locations va
LEFT JOIN bill_management_v2.vendors v ON v.code = va.vendor_code
LEFT JOIN bill_management_v2.location_detail ld ON ld.location_id = va.location_id
WHERE va.customer_id = {{ appsmith.store.selectedCustomerId || 76013 }}
ORDER BY va.location_id, va.commodity, va.id
LIMIT 5000;
