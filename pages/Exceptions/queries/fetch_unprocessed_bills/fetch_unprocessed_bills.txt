-- Unprocessed Bills: dedicated queue table for the current customer.
SELECT
    ub.id                                                 AS record_id,
    ub.fdg_id                                             AS fdg_id,
    CASE
        WHEN ub.account_codes IS NULL THEN ''
        ELSE ARRAY_TO_STRING(ub.account_codes, ', ')
    END                                                   AS service_account_id,
    ub.batch_id                                           AS batch_id,
    ub.batch_type                                         AS batch_type,
    TO_CHAR(ub.processed_at, 'MM/DD/YYYY')                AS processed_date
FROM bill_management_v2.unprocessed_bills ub
WHERE ub.customer_id = {{ appsmith.store.selectedCustomerId || 76013 }}
ORDER BY ub.id DESC
LIMIT 5000;
