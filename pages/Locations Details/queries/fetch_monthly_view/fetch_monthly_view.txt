SELECT DISTINCT ON (bm.id, bi.commodity)
    bm.id                                                        AS "Bill ID",
    br.statement_date                                            AS _sort,
    TO_CHAR(DATE_TRUNC('month', br.statement_date), 'FMMonth YYYY') AS "Month",
    UPPER(bi.commodity)                                          AS "Utility",
    btac.total_charges                                           AS "Cost",
    mu.usage                                                     AS "Usage",
    mu.uom                                                       AS "uom",
    TO_CHAR(bi.start_date, 'MM/DD/YYYY')                         AS "Start Date",
    TO_CHAR(bi.end_date, 'MM/DD/YYYY')                           AS "End Date",
    bi.days_of_service                                           AS "Days of Service"
FROM bill_management_v2.bill_metas bm
INNER JOIN bill_management_v2.bill_metas_bill_records bmbr
      ON bmbr.bill_meta_id = bm.id AND bmbr.status = 'active'
INNER JOIN bill_management_v2.bill_records br
      ON br.id = bmbr.bill_record_id
INNER JOIN bill_management_v2.bill_items bi
      ON bi.bill_record_id = br.id
INNER JOIN bill_management_v2.bill_items_virtual_accounts biva
      ON biva.bill_item_id = bi.id
LEFT JOIN bill_management_v2.bills_total_amount_charges_prior btac
      ON btac.record_id = br.id
LEFT JOIN (
    SELECT location_id, utility_type,
           DATE_TRUNC('month', time_period) AS mth,
           SUM(total_consumption)  AS usage,
           MAX(total_consumption_uom) AS uom
    FROM bill_management_v2.reports_customer_monthly_usage
    WHERE customer_id = {{ appsmith.store.selectedCustomerId }}
    GROUP BY location_id, utility_type, DATE_TRUNC('month', time_period)
) mu ON mu.location_id = biva.location_id
    AND LOWER(mu.utility_type) = LOWER(bi.commodity)
    AND mu.mth = DATE_TRUNC('month', br.statement_date)
WHERE biva.location_id = {{ (JSON.parse(appsmith.URL.queryParams.rowData || '{}').id) || 0 }}
  AND bm.customer_id = {{ appsmith.store.selectedCustomerId }}
  AND bm.deleted_at IS NULL
  AND bm.customer_ready IS NOT NULL
ORDER BY bm.id, bi.commodity, br.statement_date DESC;
