SELECT
    li.bill_id                                AS "Bill ID",
    TO_CHAR(li.statement_date, 'MM/DD/YYYY')  AS "Invoice Date",
    SUM(li.charge)                            AS "Charges"
FROM bill_management_v2.analytics_billing_line_items li
WHERE li.virtual_account_id = {{ (appsmith.store.vaSelected && appsmith.store.vaSelected.id) || 0 }}
  AND li.customer_id = {{ appsmith.store.selectedCustomerId }}
  AND li.bill_type = 'live'
  AND li.workflow_state = 'processed'
GROUP BY li.bill_id, li.statement_date
ORDER BY li.statement_date DESC;
