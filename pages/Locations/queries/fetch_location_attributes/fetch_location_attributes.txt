SELECT
    ROW_NUMBER() OVER (ORDER BY attribute_name) AS "Sort Order",
    attribute_name   AS "Name",
    attribute_format AS "Type",
    attribute_uom    AS "UoM"
FROM (
    SELECT DISTINCT attribute_name, attribute_format, attribute_uom
    FROM bill_management_v2.custom_location_attributes
    WHERE customer_id = {{ CustomerSelect.selectedOptionValue || 0 }}
) t
ORDER BY attribute_name;
