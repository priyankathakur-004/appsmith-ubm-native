{{
(function () {
	// Prepared statements are OFF for this datasource, so every value below is
	// substituted literally into the SQL string. Escape single quotes on any
	// user-supplied value to stay safe and avoid breaking on apostrophes.
	const esc = (v) => String(v).replace(/'/g, "''");
	const quoteList = (arr) => arr.map((v) => `'${esc(v)}'`).join(', ');

	const customerId = Number(CustomerSelect.selectedOptionValue) || 0;
	const search = (SearchInput.text || '').trim();
	const buildingTypes = buildingTypeFilter.selectedOptionValues || [];
	const cities = cityFilter.selectedOptionValues || [];
	const states = stateFilter.selectedOptionValues || [];

	const where = [`l.customer_id = ${customerId}`];

	if (search) {
		const like = `'%${esc(search)}%'`;
		where.push(`(
			lt.location_description ILIKE ${like} OR
			l.id::text              ILIKE ${like} OR
			lt.building_type        ILIKE ${like} OR
			lt.location_city        ILIKE ${like} OR
			lt.location_address     ILIKE ${like} OR
			lt.location_state       ILIKE ${like} OR
			lt.location_postcode    ILIKE ${like}
		)`);
	}

	if (buildingTypes.length) where.push(`lt.building_type IN (${quoteList(buildingTypes)})`);
	if (cities.length) where.push(`lt.location_city IN (${quoteList(cities)})`);
	if (states.length) where.push(`lt.location_state IN (${quoteList(states)})`);

	return `
		SELECT l.id,
		       lt.location_description AS "Description",
		       lt.building_type        AS "Type",
		       lt.location_address     AS "Service Address",
		       lt.location_city        AS "City",
		       lt.location_state       AS "State",
		       lt.location_postcode    AS "Zip",
		       lt.square_feet          AS "Area (Sq. Ft)",
		       CASE WHEN l.weather_setup THEN 'Active' ELSE 'Inactive' END AS "Weather"
		FROM bill_management_v2.locations AS l
		LEFT JOIN bill_management_v2.location_detail AS lt ON l.id = lt.location_id
		WHERE ${where.join(' AND ')}
		ORDER BY lt.location_description COLLATE "C" NULLS LAST;
	`;
})()
}}
