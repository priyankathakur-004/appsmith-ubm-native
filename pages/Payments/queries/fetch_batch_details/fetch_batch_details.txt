SELECT
    d.bill_meta_id                        AS "Bill ID",
    TO_CHAR(d.due_date, 'MM/DD/YYYY')     AS "Due Date",
    TO_CHAR(d.invoice_date, 'MM/DD/YYYY') AS "Invoice Date",
    d.vendor                              AS "Vendor",
    d.client_account                      AS "Billing ID",
    d.amount_due                          AS "Amount Due",
    d.fee                                 AS "Fees",
    d.amount_paid                         AS "Amount Paid",
    d.payment_status                      AS "Status"
FROM bill_management_v2.payment_files_logs_details_view d
WHERE d.customer_id = {{ CustomerSelect.selectedOptionValue ? parseInt(CustomerSelect.selectedOptionValue, 10) : 76013 }}
  AND d.payment_file_date::date = {{ appsmith.store.pay_active_date ? "'" + appsmith.store.pay_active_date + "'" : "NULL" }}::date
ORDER BY d.due_date;
