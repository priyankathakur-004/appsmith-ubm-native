SELECT
    d.bill_meta_id                        AS "Bill ID",
    d.amount_due                          AS "Amount Due",
    d.vendor                              AS "Vendor",
    d.client_account                      AS "Billing ID",
    TO_CHAR(d.due_date, 'YYYY-MM-DD')     AS "Due Date",
    TO_CHAR(d.invoice_date, 'YYYY-MM-DD') AS "Invoice Date",
    d.fee                                 AS "Fee",
    COALESCE(d.payment_file_type::text, 'summary') AS "File Type"
FROM bill_management_v2.payment_files_logs_details d
WHERE d.customer_id = {{ CustomerSelect.selectedOptionValue ? parseInt(CustomerSelect.selectedOptionValue, 10) : 76013 }}
  AND TO_CHAR(d.payment_file_date, 'YYYY-MM-DD') = '{{ appsmith.store.pay_dl_date || "2000-01-01" }}'
  AND COALESCE(d.payment_file_type::text, 'summary') = '{{ appsmith.store.pay_dl_type || "summary" }}'
ORDER BY d.due_date;
