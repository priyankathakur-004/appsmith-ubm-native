-- Rate Class Analysis: per-location, last-12-months electric usage + actual cost.
-- Anchored on the consumption/charges months (reports_customer_monthly_usage),
-- demand (kW) left-joined per month from the demand_billblock calculation.
-- Aggregation mirrors the Monthly Energy Consumption / Demand analytics tabs
-- (SUM across rows per location-month) so the modeled inputs match those tabs.
WITH cons AS (
    SELECT
        date_trunc('month', m.time_period)::date AS month,
        SUM(
            CASE WHEN m.bill_type = 'Supply Only'
                 THEN COALESCE(m.total_gen_consumption, 0)
                 ELSE COALESCE(m.total_consumption, 0)
            END
        ) AS kwh,
        SUM(COALESCE(m.total_charges, 0)) AS actual_charges,
        MAX(m.total_consumption_uom)      AS uom
    FROM bill_management_v2.reports_customer_monthly_usage m
    WHERE m.customer_id = {{ RC_CustomerSelect.selectedOptionValue || 76013 }}
      AND m.location_id = {{ RC_LocationSelect.selectedOptionValue || 0 }}
      AND m.utility_type = 'ELECTRIC'
    GROUP BY date_trunc('month', m.time_period)::date
),
dem AS (
    SELECT
        date_trunc('month', br.statement_date)::date AS month,
        SUM(CASE WHEN c.name = 'demand_billblock' THEN c.value ELSE 0 END) AS kw
    FROM bill_management_v2.bill_metas bm
    INNER JOIN bill_management_v2.bill_metas_bill_records bmbr ON bmbr.bill_meta_id = bm.id AND bmbr.status = 'active'
    INNER JOIN bill_management_v2.bill_records br ON br.id = bmbr.bill_record_id
    INNER JOIN bill_management_v2.bill_items bi ON bi.bill_record_id = br.id
    INNER JOIN bill_management_v2.bill_items_virtual_accounts biva ON biva.bill_item_id = bi.id
    LEFT JOIN bill_management_v2.calculations c ON c.bill_item_id = bi.id
    WHERE bm.deleted_at IS NULL AND bm.customer_ready IS NOT NULL
      AND bm.customer_id = {{ RC_CustomerSelect.selectedOptionValue || 76013 }}
      AND biva.location_id = {{ RC_LocationSelect.selectedOptionValue || 0 }}
      AND bi.commodity = 'ELECTRIC'
    GROUP BY date_trunc('month', br.statement_date)::date
)
SELECT
    TO_CHAR(cons.month, 'YYYY-MM-DD')      AS month,
    ROUND(cons.kwh::numeric, 2)            AS kwh,
    ROUND(COALESCE(dem.kw, 0)::numeric, 2) AS kw,
    ROUND(cons.actual_charges::numeric, 2) AS actual_charges,
    cons.uom,
    loc.postcode,
    loc.state,
    loc.country,
    loc.name                               AS location_name
FROM cons
LEFT JOIN dem ON dem.month = cons.month
CROSS JOIN (
    SELECT postcode, name, state, country
    FROM bill_management_v2.locations
    WHERE id = {{ RC_LocationSelect.selectedOptionValue || 0 }}
) loc
ORDER BY cons.month DESC
LIMIT 12;
