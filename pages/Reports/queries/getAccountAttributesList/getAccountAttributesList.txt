-- Account / VA (Virtual Account) attribute names for the picker.
-- Mirrors getLocationAttributesList. UBM "account attributes" = VA attributes.
-- Schema confirmed 2026-06-17: attribute *names* live in
-- virtual_accounts_attributes_metadata (per-customer, soft-deletable). The
-- actual VA->value rows are in virtual_accounts_attributes_mapping
-- (value-level filtering, second-level picker, not built yet).
SELECT DISTINCT attribute_name AS value, attribute_name AS label
FROM bill_management_v2.virtual_accounts_attributes_metadata
WHERE customer_id = {{ Number(CustomerSelect.selectedOptionValue) || 94512 }}
  AND deleted_at IS NULL
  AND attribute_name IS NOT NULL AND attribute_name <> ''
ORDER BY attribute_name;
