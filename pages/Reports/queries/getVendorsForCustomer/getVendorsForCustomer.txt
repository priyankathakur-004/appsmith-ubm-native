SELECT DISTINCT amf.vendor_code AS value,
  COALESCE(cvn.pretty_name, amf.vendor_code) AS label
FROM bill_management_v2.analytics_monthly_feed amf
LEFT JOIN bill_management_v2.customers_providers_pretty_name cvn
  ON cvn.code = amf.vendor_code AND cvn.customer_id = amf.customer_id
WHERE amf.customer_id = {{ ReportSpecs.customerId() }}
  AND amf.vendor_code IS NOT NULL AND amf.vendor_code <> ''
ORDER BY label;
