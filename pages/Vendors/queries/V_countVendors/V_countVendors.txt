{{
(function() {
	const s = (V_SearchInput.text || '').trim().replace(/'/g, "''");
	const prov = (V_ProviderInput.text || '').trim().replace(/'/g, "''");
	const blankOnly = V_BlankToggle.isSwitchedOn;
	let q = `SELECT COUNT(*) AS count FROM bill_management_v2.vendors v WHERE 1=1`;
	if (s !== '') {
		q += ` AND (CAST(v.id AS text) ILIKE '%${s}%' OR v.code ILIKE '%${s}%' OR COALESCE(v.pretty_name, '') ILIKE '%${s}%')`;
	}
	if (blankOnly) {
		q += ` AND (v.pretty_name IS NULL OR v.pretty_name = '')`;
	}
	if (prov !== '') {
		q += ` AND EXISTS (SELECT 1 FROM bill_management_v2.providers_vendors pv WHERE pv.vendor_id = v.id AND CAST(pv.provider_id AS text) ILIKE '%${prov}%')`;
	}
	return q;
})()
}}
